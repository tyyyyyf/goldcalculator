function renderItin() { const list = document.getElementById('itinerary-list'); list.innerHTML = ""; const items = window.state.itin[window.state.currentDay] || []; items.forEach((item, idx) => { const color = getItinColor(item.type); const div = document.createElement('div'); div.className = `itin-card ${color.bg} ${color.text} p-6 rounded-[2.5rem] shadow-sm relative transition-all`; div.setAttribute('data-idx', idx); const addressLine = (item.addrKr || item.addrCn) ? `

📍 ${item.addrKr || item.addrCn}

` : ''; const descContent = item.desc ? `
${urlify(item.desc)}
` : ''; div.innerHTML = `
${item.time}

${item.title || '地點'}

${addressLine}
${item.type || '其他'}
${descContent} ${item.img ? `` : ''}
`; list.appendChild(div); }); initSortable(); }
